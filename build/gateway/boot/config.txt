# Raspberry Pi 4 boot configuration — Gateway node

[pi4]
kernel=kernel.img
arm_freq=1800
gpu_mem=256
enable_gic=1
dtoverlay=disable-bt
dtparam=i2c_arm=on
dtparam=spi=on
# Load compiled braincraft overlay (braincraft.dtbo must be in /boot/overlays/)
# Fan control is handled by the braincraft overlay's gpio-fan node; no separate dtoverlay=gpio-fan needed.
dtoverlay=braincraft
enable_uart=1
